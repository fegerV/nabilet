# robots.txt — NABILET
#
# ПРАВИЛО: файл пишется вместе со страницами, а не раньше.
# Каждый `Disallow` ниже закрывает раздел, который реально существует
# и не должен попадать в индекс. Если раздел удалён — удалите строку;
# если добавлен служебный — добавьте. «На всякий случай» запрещать
# существующие страницы нельзя: закрытая роботом полезная страница
# теряет и трафик, и внешние ссылки (Google не может их перераспределить).
#
# Чего здесь СОЗНАТЕЛЬНО нет:
#   /event/*  — это и есть SEO-страницы мероприятий, они должны индексироваться
#   /         — витрина (каталог)
#   /assets/  — статика Vite, нужна для отрисовки. Запрет не экономит
#               краулинговый бюджет, но ломает рендеринг постраничных превью.

User-agent: *
Allow: /

# Админ-панель. Токен в localStorage, но URL-ы утекают через логи рефереров
# и скриншоты; индексировать их незачем.
Disallow: /admin
Disallow: /admin/

# Оформление заказа и оплата. Страницы живут на персональных данных
# (корзина, номер заказа, сумма) и не имеют уникального контента.
Disallow: /checkout
Disallow: /checkout/
Disallow: /payment/
Disallow: /cart

# Выпущенные билеты. GET /api/v1/my-tickets намеренно вне auth:api —
# скоупится по заголовку X-Cart-Token. Это публичный по дизайну эндпоинт,
# и его URL не должен попасть в индекс ни при каких условиях.
Disallow: /tickets
Disallow: /api/v1/tickets
Disallow: /api/v1/my-tickets
Disallow: /api/v1/me/

# Установщик. Одноразовый pre-auth мастер; после создания storage/install.lock
# отдаёт 410, но до установки он открыт и не должен индексироваться.
Disallow: /install
Disallow: /install/

# API целиком. Отдаёт JSON (не HTML), индексация бессмысленна, а ответы
# с ошибками (§66 envelope) в индексе выглядят как «мусорный» контент.
# ИСКЛЮЧЕНИЕ: sitemap.xml живёт ВНУТРИ /api/v1 (роуты модуля Seo подключены
# централизованно из routes/api.php) и его запрещать нельзя ни в коем случае —
# иначе поисковик не сможет прочитать карту сайта.
Disallow: /api/v1/

# Карта сайта. Путь именно такой: модуль Seo подключён в API-группу,
# поэтому схема доступна на /api/v1/sitemap.xml, а не на /sitemap.xml.
# Если роуты когда-нибудь переедут на корень — эту строку надо поменять,
# иначе robots.txt будет указывать на 404 и краулер это запишет в отчёт.
Sitemap: /api/v1/sitemap.xml
